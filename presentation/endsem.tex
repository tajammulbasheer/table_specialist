\begin{frame}{The problem: specialists need labels}
\vspace{0.0cm}
\begin{columns}[T]
\column{0.32\textwidth}
\begin{tscard}[tsmuted]{Option 1}{Prompt a generalist}
No training, any task.\\[2pt]
\textbf{But} weak on tables. Vanilla Llama-3.1-8B-Instruct, zero-shot:\\[2pt]
NL-to-SQL \textbf{0.50}\\ NL-to-Scala \textbf{0.19}\\ Error detection F1 \textbf{0.09}
\end{tscard}
\column{0.32\textwidth}
\begin{tscard}[tsnavy]{Option 2}{Fine-tune a specialist}
One model per task: more accurate, and an 8B model is cheap to run.\\[2pt]
\textbf{But} it needs labelled data for every task.
\end{tscard}
\column{0.32\textwidth}
\begin{tscard}[tsred]{The catch}{Table labels are costly}
A label is a \textbf{program} (SQL, Scala, R), a set of cells or a column mapping.\\[2pt]
Experts must write and check each one.
\end{tscard}
\end{columns}
\vspace{0.2cm}
\begin{tscallout}[tspink]
\large Can an LLM become a table specialist \textbf{without any human labels}?
\end{tscallout}
\end{frame}

\begin{frame}{Key idea: a dual task checks every answer}
\vspace{0.05cm}
{\small Agreement between two independent views \textbf{replaces the human label}.}\par\vspace{0.08cm}
\begin{columns}[T]
\column{0.485\textwidth}
\begin{tscard}[tsnavy]{Generative tasks}{Output is a program}
\textbf{Examples:} NL-to-SQL, NL-to-Scala, NL-to-R.\\[2pt]
\textbf{Dual:} the same answer in a \textbf{second language}.\\[2pt]
\textbf{Check:} run both; keep only if the results \textbf{match}.

\begin{frame}{Generative task: checked by execution}
\vspace{0.05cm}
\begin{columns}[T]
\column{0.36\textwidth}
{\small\textbf{\color{tsnavy}Table} \muted{(one of 3,000)}}\par\vspace{0.08cm}
{\footnotesize\setlength{\tabcolsep}{5pt}\renewcommand{\arraystretch}{1.0}
\begin{tabular}{@{}lll@{}}
\toprule
\thd{Athlete} & \thd{Country} & \thd{Medals}\\\midrule
Ana & ESP & 3\\ Ben & USA & 5\\ Chen & CHN & 4\\ Dana & USA & 2\\
\bottomrule
\end{tabular}}
\par\vspace{0.2cm}
{\small\textbf{\color{tsnavy}Question} \muted{(written by the model)}}\par
{\small ``How many medals did athletes from the USA win in total?''}
\column{0.6\textwidth}
\begin{tscard}[tsnavy]{Two programs, one question}{SQL and Spark-Scala}
\textbf{SQL}\enspace\code{SELECT SUM(Medals) FROM t WHERE Country='USA'}\\[3pt]
\textbf{Scala}\enspace\code{df.filter(\$"Country"==="USA").agg(sum("Medals"))}
\end{tscard}
\vspace{0.05cm}
{\footnotesize\setlength{\tabcolsep}{4pt}\renewcommand{\arraystretch}{1.0}
\begin{tabular}{@{}lccccc@{\hspace{10pt}}l@{}}
\thd{Sample} & $S_1$ & $S_2$ & $S_3$ & \ldots & $S_{10}$ & \\\midrule
SQL & 7 & 5 & 2 & \ldots & 7 & \\
Scala & 7 & 5 & 2 & \ldots & 7 & \\
& \ok & \ok & \ok & & \ok & $\Rightarrow$ \textbf{\color{tsgreen}keep the pair}\\
\end{tabular}}
\end{columns}
\vspace{0.2cm}
\begin{tscallout}[tspanel]\small
Had the Scala \emph{counted} rows instead of summing medals, it would return 2, not 7 $\Rightarrow$ \textbf{\color{tsred}reject}.
Two wrong programs rarely agree on \textbf{ten} random halves of a table.
\end{tscallout}
\vspace{-0.05cm}
\muted{\scriptsize Illustrative example. Each sample $S_k$ is a random 50\% of the rows; both programs run on the same sample.}
\end{frame}

\begin{frame}{Classification task: checked by shuffling}
\vspace{0.05cm}
\begin{columns}[T]
\column{0.36\textwidth}
\begin{tscard}[tsred]{Step 1: generate}{Inject a known error}
Clean column \code{City}: Paris, London, Berlin, Rome.\\[2pt]
The model writes an error, e.g.\ the typo \textbf{Lodnon} (or a semantic error such as \textbf{Tuesday}).
\end{tscard}
\column{0.6\textwidth}
{\small\textbf{\color{tsnavy}Step 2: validate}\enspace 5 shuffled orders}\par\vspace{0.1cm}
{\footnotesize\setlength{\tabcolsep}{5pt}\renewcommand{\arraystretch}{1.05}
\begin{tabular}{@{}cll@{}}
\toprule
\thd{Order} & \thd{Values shown} & \thd{Answer}\\\midrule
1 & Rome, \textbf{Lodnon}, Paris, Berlin, London & Lodnon \ok\\
2 & London, Berlin, Rome, \textbf{Lodnon}, Paris & Lodnon \ok\\
3 & \textbf{Lodnon}, Paris, London, Rome, Berlin & Lodnon \ok\\
4 & Paris, Rome, Berlin, London, \textbf{Lodnon} & Lodnon \ok\\
5 & Berlin, \textbf{Lodnon}, Rome, London, Paris & Lodnon \ok\\
\bottomrule
\end{tabular}}
\par\vspace{0.12cm}
{\small 5 of 5 $\Rightarrow$ \textbf{\color{tsgreen}keep}, labelled ``Lodnon''.}\par\vspace{0.05cm}
{\footnotesize\muted{Why shuffle? A validator that always names the last value would agree by luck. Clean columns are kept only if \textbf{nothing} is named in all 5 orders.}}
\end{columns}
\vspace{0.1cm}
\muted{\scriptsize Illustrative example. The repository uses 5 fixed shuffle seeds (0--4) and separate prompts for typos and semantic errors.}
\end{frame}

\begin{frame}{Why each part of the design is there}
\vspace{0.1cm}
\begin{columns}[T]
\column{0.485\textwidth}
\begin{tscard}[tsnavy]{Dual check}{The only source of supervision}
Without it, the model trains on its own mistakes. The paper's ``no validation'' ablation tests exactly this, and so do we.
\end{tscard}
\vspace{0.05cm}
\begin{tscard}[tsnavy]{10 random samples}{Guards against chance agreement}
Two wrong programs can agree on one table, but rarely on ten random halves.
\end{tscard}
\column{0.485\textwidth}
\begin{tscard}[tsred]{Permutations}{Guards against position bias}
The classifier must name the same value whatever the order.
\end{tscard}
\vspace{0.05cm}
\begin{tscard}[tsred]{Rounds}{Better model, better data}
Round $r$'s model writes round $r{+}1$'s data, and a better validator keeps more of it. The paper restarts LoRA from vanilla each round.
\end{tscard}
\end{columns}
\vspace{0.25cm}
\begin{tscallout}[tspanel]\small The price: \textbf{strict agreement throws most generated data away}. That is where our extension starts.\end{tscallout}
\end{frame}

\item Numbers equal within $10^{-6}$; true/false = 1/0
\item All-empty rows dropped

\begin{frame}{Repair-TS in detail}
\vspace{0.05cm}
\begin{columns}[T]
\column{0.485\textwidth}
\begin{tscard}[tsnavy]{1. Which side is repaired}{Set by the failure type}
\begin{itemize}\setlength{\itemsep}{0pt}
\item SQL fails or always empty: \textbf{SQL}
\item Scala fails: \textbf{Scala}
\item They disagree: \textbf{both}, then try new+old, old+new, new+new
\end{itemize}
\end{tscard}
\vspace{0.05cm}
\begin{tscard}[tsnavy]{2. Feedback}{Only from its own run}
Its error, ``returns no rows'', or its first 5 rows on the failing sample.
\end{tscard}
\column{0.485\textwidth}
\begin{tscard}[tsred]{3. The check stays honest}{Independence is kept}
A program \textbf{never} sees the other language's code or output, so agreement still means two independent programs agree.
\end{tscard}
\vspace{0.05cm}
\begin{tscard}[tsred]{4. One attempt}{Greedy, then $\leq$\,3 samples}
Samples only if the reply is empty or unchanged. Also used for NL-to-R and error detection.
\end{tscard}
\end{columns}
\end{frame}

\begin{frame}{Summary}
\vspace{0.15cm}
\begin{enumerate}\setlength{\itemsep}{0.45em}
\item \textbf{Reproduction:} Instruct TS scores 0.6113 against the paper's 0.5983 --- on Instruct, not the README's base model.
\item \textbf{All four tasks:} TS beats vanilla on SQL, Scala, R and error detection; Scala and errors land above the paper's figures.
\item \textbf{Validation:} pays off over rounds, not in round 1.
\item \textbf{Rejected data:} DPO hurts and resampling barely helps; \alert{repair-TS} clearly helps base SQL, Scala and R, and is never worse.
\item \textbf{Controls:} on base SQL, repaired data is worth more than as much fresh data.
\end{enumerate}
